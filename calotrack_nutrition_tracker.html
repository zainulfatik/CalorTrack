Pengaturan Integration Google Apps Script

Salin kode Apps Script di bawah ini ke Google Sheets Anda agar terhubung secara otomatis:

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var dateObj = new Date(data.tanggal);
    
    // Nama sheet: Bulan Tahun (cth: September 2026)
    var monthNames = ["Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember"];
    var sheetName = monthNames[dateObj.getMonth()] + " " + dateObj.getFullYear();
    
    var sheet = ss.getSheetByName(sheetName);
    
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(["Tanggal", "Asupan Kalori", "Kebutuhan Kalori", "Protein", "Defisit/Surplus", "Berat Badan"]);
      sheet.getRange("A1:F1").setFontWeight("bold").setBackground("#d9ead3");
    }
    
    var deficitSurplus = (data.kalori && data.kalori > 0) ? (data.kalori - 1630) : "";
    
    if (data.type === "daily_summary") {
       sheet.appendRow([data.tanggal, data.kalori, 1630, data.protein, deficitSurplus, data.berat_badan]);
    } else if (data.type === "weight_update") {
       sheet.appendRow([data.tanggal, "", "", "", "", data.berat_badan]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({"error": error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
